using System;
using System.Collections.Generic;
namespace Student
{
    class Student
    {
        public int id;
        public string Name;
        public int Age;
        public string Course;
        public double Marks;
    }
    class Program
    {
        static void Main(String[] args)
        {
            List<Student>students=new List<Student>();
            students.Add(new Student
            {
                id=1,
                Name="Ayush-raj",
                Age=22,
                Course="CSE",
                Marks=100
            });
            students.Add(new Student
            {
                id=2,
                Name="John",
                Age=20,
                Course="IT",
                Marks=90
            });
            foreach(Student student in students)
            {
                Console.WriteLine(
                    student.Name+" - "+
                    student.Age+" - "+
                    student.Course+" - "+
                    student.Marks   
                );
            }
            Console.ReadLine();
        }
    }
}
